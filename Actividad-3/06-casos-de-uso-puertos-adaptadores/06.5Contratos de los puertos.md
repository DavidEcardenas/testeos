public interface IDeliveryRepository
{
    Task<Delivery?> GetAsync(DeliveryId id, CancellationToken ct);
    Task SaveAsync(Delivery delivery, CancellationToken ct);
}

public interface ICourierAvailabilityPort
{
    Task<bool> IsAvailableAsync(CourierId id, CancellationToken ct);
}

public interface IRouteCapacityPort
{
    Task<bool> HasCapacityAsync(CourierId id, CancellationToken ct);
}

public interface IStockReservationPort
{
    Task<bool> IsReservedAsync(OrderId orderId, CancellationToken ct);
}

public interface IEventPublisher
{
    Task PublishAsync<TEvent>(TEvent @event, CancellationToken ct)
        where TEvent : class;
}
