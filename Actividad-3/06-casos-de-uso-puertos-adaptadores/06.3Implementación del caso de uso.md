public sealed class AssignCourierUseCase
{
    private readonly IDeliveryRepository _deliveryRepo;
    private readonly ICourierAvailabilityPort _courierPort;
    private readonly IRouteCapacityPort _routePort;
    private readonly IStockReservationPort _stockPort;
    private readonly IEventPublisher _events;

    public AssignCourierUseCase(
        IDeliveryRepository deliveryRepo,
        ICourierAvailabilityPort courierPort,
        IRouteCapacityPort routePort,
        IStockReservationPort stockPort,
        IEventPublisher events)
    {
        _deliveryRepo = deliveryRepo;
        _courierPort = courierPort;
        _routePort = routePort;
        _stockPort = stockPort;
        _events = events;
    }

    public async Task<AssignCourierResult> ExecuteAsync(
        AssignCourierCommand cmd, CancellationToken ct)
    {
        var delivery = await _deliveryRepo.GetAsync(
            new DeliveryId(cmd.DeliveryId), ct)
            ?? throw new DomainException("Delivery not found");

        // INV-02: stock reservado
        if (!await _stockPort.IsReservedAsync(delivery.OrderId, ct))
            throw new DomainException("Stock not reserved");

        // INV-04: disponibilidad
        if (!await _courierPort.IsAvailableAsync(
                new CourierId(cmd.CourierId), ct))
            throw new DomainException("Courier not available");

        // INV-03: capacidad de ruta
        if (!await _routePort.HasCapacityAsync(
                new CourierId(cmd.CourierId), ct))
            throw new DomainException("Route capacity exceeded");

        // INV-01 e INV-06: la entidad protege las invariantes
        delivery.AssignCourier(new CourierId(cmd.CourierId), cmd.RequestedBy);

        await _deliveryRepo.SaveAsync(delivery, ct);

        await _events.PublishAsync(
            new DeliveryAssigned(
                delivery.Id.Value,
                cmd.CourierId,
                DateTime.UtcNow),
            ct);

        return new AssignCourierResult(
            delivery.Id.Value,
            cmd.CourierId,
            DateTime.UtcNow);
    }
}
